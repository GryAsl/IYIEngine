# Gate 1: raylib + WebGL 2 smoke, same raylib as the engine. Build with emcmake (see ../../README.md).
cmake_minimum_required(VERSION 3.20)
project(iyi_web_smoke_raylib CXX C)
set(CMAKE_CXX_STANDARD 20)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PLATFORM Web CACHE STRING "" FORCE)
set(GRAPHICS GRAPHICS_API_OPENGL_ES3 CACHE STRING "" FORCE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 6.2 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(raylib)
add_executable(smoke main.cpp)
target_link_libraries(smoke raylib)
set_target_properties(smoke PROPERTIES SUFFIX ".html")
target_link_options(smoke PRIVATE -sUSE_GLFW=3 -sMIN_WEBGL_VERSION=2 -sMAX_WEBGL_VERSION=2 -sALLOW_MEMORY_GROWTH=1)
